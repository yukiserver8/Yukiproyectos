7. GENERADOR DE CRC32
============================================================

Para calcular el CRC32 de un archivo desde PowerShell:

=== OPCIÓN A: Funci�n completa (recomendada) ===

function Get-CRC32Safe([string]$path) {
    $table = New-Object System.UInt32[] 256;
    for ($i = 0; $i -lt 256; $i++) {
        $c = [System.UInt32]$i;
        for ($j = 0; $j -lt 8; $j++) {
            if (($c -band 1) -ne 0) {
                $c = [System.UInt32](0xEDB88320 -bxor ($c -shr 1))
            } else {
                $c = [System.UInt32]($c -shr 1)
            }
        };
        $table[$i] = $c
    };
    $crc = [System.UInt32]::MaxValue;
    $buffer = [System.IO.File]::ReadAllBytes($path);
    for ($i = 0; $i -lt $buffer.Length; $i++) {
        $crc = [System.UInt32](($crc -shr 8) -bxor $table[($crc -band 0xFF) -bxor $buffer[$i]])
    };
    $crc = [System.UInt32]($crc -bxor [System.UInt32]::MaxValue);
    $bytes = [System.BitConverter]::GetBytes($crc);
    if ([System.BitConverter]::IsLittleEndian) { [Array]::Reverse($bytes) };
    return ($bytes | ForEach-Object { "{0:x2}" -f $_ }) -join ''
}
